import "dotenv/config";
import express from "express";
import { z } from "zod";
import { publishBurn } from "./engine.js";

const app = express();
app.use(express.json({ limit: "32kb" }));

app.get("/healthz", (_req, res) => {
  res.json({ ok: true, service: "stream-burn-publisher" });
});

app.get("/readyz", (_req, res) => {
  const ready = Boolean(
    process.env.SOLANA_RPC_URL &&
    process.env.STREAM_MINT &&
    process.env.X_ACCESS_TOKEN
  );

  res.status(ready ? 200 : 503).json({ ready });
});

app.post("/publish", async (req, res) => {
  const parsed = z.object({
    signature: z.string().min(32).max(100)
  }).safeParse(req.body);

  if (!parsed.success) {
    return res.status(400).json({
      error: "signature is required"
    });
  }

  try {
    const result = await publishBurn(parsed.data.signature);
    return res.json(result);
  } catch (error) {
    const message = error instanceof Error ? error.message : "Unknown error";
    return res.status(422).json({ error: message });
  }
});

const port = Number(process.env.PORT ?? 8080);
app.listen(port, () => {
  console.log(`STREAM burn publisher listening on :${port}`);
});
